===== B3 : handleWorkerFinished -- worker terminal result delivery + resultDisposed + jobDead + direct-reap attempt + tryFinalize  (archiveextractionmanager.cpp:347-437) =====
347: void ArchiveExtractionManager::handleWorkerFinished(int handle,
348:                                                     ArchiveExtractionWorker::ResultType result,
349:                                                     const QStringList &files,
350:                                                     const QString &errorMessage)
351: {
352:     // Must be called from manager's thread (main thread)
353:     Q_ASSERT(QThread::currentThread() == thread());
354: 
355:     std::shared_ptr<ExtractionEntry> entry;
356:     {
357:         QMutexLocker locker(&m_entriesMutex);
358:         if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
359:             entry = m_entries[handle];
360:         }
361:     }
362: 
363:     if (!entry) {
364:         return; // Already reaped/finalized
365:     }
366: 
367:     // Reentrancy guard: duplicate or reentrant result notifications are harmless.
368:     if (entry->resultClaimed.exchange(true, std::memory_order_acq_rel)) {
369:         return;
370:     }
371: 
372:     // Capture QPointer before dispatching; never dereference after a handler may emit result.
373:     QPointer<ArchiveExtractionJob> job = entry->job;
374: 
375:     // The worker's emitted result is authoritative. Do NOT override Success/FallbackRequired
376:     // with Cancelled based on a late-arriving cancellation flag.
377:     if (job) {
378:         if (entry->job->testWorkerResultCallback()) {
379:             entry->job->testWorkerResultCallback()(result);
380:         }
381:         switch (result) {
382:             case ArchiveExtractionWorker::Success:
383:                 entry->job->handleSuccess(files);
384:                 break;
385:             case ArchiveExtractionWorker::FallbackRequired:
386:                 entry->job->handleFallbackRequired();
387:                 break;
388:             case ArchiveExtractionWorker::ExtractionFailed:
389:                 entry->job->handleExtractionFailed(errorMessage);
390:                 break;
391:             case ArchiveExtractionWorker::Cancelled:
392:                 entry->job->handleCancelled();
393:                 break;
394:         }
395:     } else {
396:         if (entry->testWorkerResultCallback) {
397:             entry->testWorkerResultCallback(result);
398:         }
399:     }
400: 
401:     // Result disposition is complete from the manager's perspective once handlers ran.
402:     entry->resultDisposed.store(true, std::memory_order_release);
403: 
404:     if (!job) {
405:         // Job object already destroyed before result delivery. No live consumer remains.
406:         entry->jobDead.store(true, std::memory_order_release);
407:         // If the thread is already reaped, clean up any unconsumed payload now.
408:         // Otherwise reapEntry will handle it once threadReaped becomes true.
409:         if (entry->threadReaped.load(std::memory_order_acquire)) {
410:             cleanupPayloadIfNeeded(entry);
411:         }
412:     }
413: 
414:     // Attempt direct reaping if the thread has already finished.
415:     QThread *workerThread = entry->thread;
416:     if (workerThread && !entry->threadReaped.load(std::memory_order_acquire)) {
417:         if (!workerThread->isRunning() && workerThread->wait(0)) {
418:             delete workerThread;
419:             entry->thread = nullptr;
420:             entry->worker = nullptr;
421:             entry->threadReaped.store(true, std::memory_order_release);
422:             // If the job is dead, clean up any unconsumed payload now that
423:             // both result delivery and physical reaping are complete.
424:             if (entry->jobDead.load(std::memory_order_acquire)) {
425:                 cleanupPayloadIfNeeded(entry);
426:             }
427:         } else if (workerThread->isRunning()) {
428:             entry->state.store(ExtractionEntry::State::Finishing, std::memory_order_release);
429:             scheduleReapRetry(handle);
430:             return;
431:         }
432:     }
433: 
434:     // Physical reaping happens independently via reapEntry (scheduled by thread.finished).
435:     // Finalize only when both resultDisposed && threadReaped are true.
436:     tryFinalize(entry);
437: }
