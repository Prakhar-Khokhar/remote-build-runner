+        qCritical() << "Active extractions count:" << activeCount;
+        qCritical() << "Configuration has been restored, but extraction workers may still be running.";
+        qCritical() << "Terminating test process to avoid unsafe exit with active workers.";
+        std::quick_exit(EXIT_FAILURE);
+    }
+    // Do not clear destination mutexes here - mutexes are process-lifetime and
+    // clearing them while worker threads might still be running is unsafe.

ArchiveExtractionJob::ArchiveExtractionJob(const QString &payloadFile, const QString &installDir, UncompressionOption uncompressionOption, QObject *parent)
    : KJob(parent)
    , d(new ArchiveExtractionJobPrivate)
{
    d->payloadFile = payloadFile;
    d->installDir = installDir;
    d->uncompressionOption = uncompressionOption;
    d->cancellation = std::make_shared<std::atomic<bool>>(false);
}

ArchiveExtractionJob::~ArchiveExtractionJob()
{
    // If kill was requested but start was never called, clean up the payload
    if (d->killRequested && !d->startCalled) {
        QFile::remove(d->payloadFile);
    }

    // If the job hasn't finished yet (destroyed while worker is running),
    // request cancellation. If the job already finished, the manager has
    // already cleaned up via QThread::finished → reapEntry.
    if (d->extractionHandle != -1 && !d->finished) {
        d->cancellation->store(true, std::memory_order_release);
        ArchiveExtractionManager::instance().requestCancellation(d->extractionHandle);
    }
}

void ArchiveExtractionJob::start()
{
    if (d->startCalled) {
        return;
    }
    d->startCalled = true;

    // If kill was requested before start, finish immediately with KilledJobError
    // and clean up the payload since no worker will be created to handle it.
    if (d->killRequested) {
        QFile::remove(d->payloadFile);
        if (!error()) {
            setError(KJob::KilledJobError);
            setErrorText(i18n("Archive extraction was cancelled"));
            emitResult();
        } else if (error() == KJob::KilledJobError && errorText().isEmpty()) {
            setErrorText(i18n("Archive extraction was cancelled"));
        }
        return;
    }

    // Start extraction through the manager
    d->extractionHandle = ArchiveExtractionManager::instance().startExtraction(

bool ArchiveExtractionJob::doKill()
{
    d->cancellation->store(true, std::memory_order_release);
    d->killRequested = true;

    // If already finished, nothing to do
    if (d->finished) {
        return true;
    }

    // If extraction hasn't started yet, we can finish immediately
    if (d->extractionHandle == -1) {
        return true;
    }

    // Request cancellation through manager
    ArchiveExtractionManager::instance().requestCancellation(d->extractionHandle);

    // Return false - job will finish when worker completes and emits result
    return false;
}

int ArchiveExtractionManager::startExtraction(const QString &payloadFile,
                                              const QString &installDir,
                                              ArchiveExtractionJob::UncompressionOption uncompressionOption,
                                              ArchiveExtractionJob *job)
{
    // Must be called from manager's thread (main thread)
    Q_ASSERT(QThread::currentThread() == thread());

    const int handle = m_nextHandle.fetch_add(1, std::memory_order_relaxed);

    auto entry = std::make_shared<ExtractionEntry>();
    entry->payloadFile = payloadFile;
    entry->installDir = installDir;
    entry->uncompressionOption = uncompressionOption;
    entry->job = job;
    entry->handle = handle;
    entry->state.store(ExtractionEntry::State::Starting, std::memory_order_release);

    // Share the job's cancellation flag with the manager entry
    if (job) {
        entry->cancellation = job->cancellationFlag();
    } else {
        entry->cancellation = std::make_shared<std::atomic<bool>>(false);
    }

    // Store entry before starting thread (so cancellation/job destruction can find it)
    {
        QMutexLocker locker(&m_entriesMutex);
        if (static_cast<size_t>(handle) >= m_entries.size()) {
            m_entries.resize(handle + 1);
        }
        m_entries[handle] = entry;
    }

    ExtractionEntry *storedEntry = entry.get();
    Q_ASSERT(storedEntry);

    // Install job-destruction observation BEFORE any result delivery.
    // This ensures that even if startExtraction() synchronously emits a
    // result (e.g., synthetic failure), the manager can track job destruction
    // and release the entry once both notifications are complete.
    if (storedEntry->job) {
        QObject::connect(storedEntry->job, &QObject::destroyed,
                         this, [this, handle]() {
            QMetaObject::invokeMethod(this, [this, handle]() {
                std::shared_ptr<ExtractionEntry> e;
                {
                    QMutexLocker locker(&m_entriesMutex);
                    if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
                        e = m_entries[handle];
                    }
                }
                if (!e) {
                    return;
                }
                e->jobDead.store(true, std::memory_order_release);
                if (e->resultDisposed.load(std::memory_order_acquire)
                    && e->threadReaped.load(std::memory_order_acquire)) {
                    cleanupPayloadIfNeeded(e);
                }
                tryFinalize(e);
            }, Qt::QueuedConnection);
        }, Qt::DirectConnection);
    }

    // Set the extraction handle on the job BEFORE any result delivery.
    // If startExtraction() synchronously emits a result (e.g., synthetic
    // failure) and the result handler destroys the job, the job destructor
    // will see d->finished == true (set before emitResult in handleExtractionFailed)
    // and will skip requestCancellation, leaving entry finalization to
    // deliverSyntheticFailure -> tryFinalize, which is already in progress.
    // The job-destruction connection installed above ensures the manager
    // observes job destruction and releases the entry.
    // Set the extraction handle on the job BEFORE any result delivery.
    // If startExtraction() synchronously emits a result (e.g., synthetic
    // failure) and the result handler destroys the job, the job will have a
    // valid handle. The job destructor checks d->finished (set before
    // emitResult in handleExtractionFailed) and skips requestCancellation,
    // leaving entry finalization to deliverSyntheticFailure -> tryFinalize.
    // The job-destruction connection installed above ensures the manager
    // observes job destruction and releases the entry.
    if (storedEntry->job) {
        storedEntry->job->setExtractionHandle(handle);
    }

    // Corrected startup sequence: start the thread BEFORE creating/moving the worker.
    // This avoids moving a worker into a thread whose native creation failed.
    QThread *newThread = new QThread();
    entry->thread = newThread;

    bool started = false;
    if (m_testForceStartFailure) {
        // Deterministic test seam: simulate a failed native thread start.
        started = false;
    } else {
        newThread->start();
        started = newThread->isRunning();
    }

    if (!started) {
        // Failed-start path: no worker was ever created or moved, so no consumer
        // can claim the payload. Clean it up directly, finalize the entry, and
        // report a synthetic terminal extraction failure.
        delete newThread;
        entry->thread = nullptr;
        entry->state.store(ExtractionEntry::State::Done, std::memory_order_release);
        entry->threadReaped.store(true, std::memory_order_release);

        if (!entry->payloadFile.isEmpty() && QFileInfo::exists(entry->payloadFile)) {
            if (!QFile::remove(entry->payloadFile)) {
                qCWarning(KNEWSTUFFCORE) << "Failed to remove payload" << entry->payloadFile
                                         << "after thread start failure";
            }
        }

        deliverSyntheticFailure(entry, QStringLiteral("Archive extraction thread could not be started"));
        return handle;
    }

    // Thread started successfully.

    // Create the worker on the manager thread.
    entry->worker = new ArchiveExtractionWorker(payloadFile, installDir,
        static_cast<ArchiveExtractionWorker::UncompressionOption>(uncompressionOption),
        entry->cancellation);

    if (job) {
        if (auto b = job->testBarrierAfterStarted()) {
            entry->worker->setBarrierAfterStarted(b);
        }
        if (auto b = job->testBarrierBeforeCopyTo()) {
            entry->worker->setBarrierBeforeCopyTo(b);
        }
        if (auto b = job->testBarrierBeforeCancellationCheck()) {
            entry->worker->setBarrierBeforeCancellationCheck(b);
        }
        if (job->testWorkerStartedCallback()) {
            entry->worker->setTestStartedCallback(job->testWorkerStartedCallback());
        }
        if (job->testBeforeCopyToCallback()) {
            entry->worker->setTestBeforeCopyToCallback(job->testBeforeCopyToCallback());
        }
        if (job->testBeforeCancellationCheckCallback()) {
            entry->worker->setTestBeforeCancellationCheckCallback(job->testBeforeCancellationCheckCallback());
        }
        if (job->testAboutToCopyToCallback()) {
            entry->worker->setTestAboutToCopyToCallback(job->testAboutToCopyToCallback());
        }
        if (job->testPreCopyToCallback()) {
            entry->worker->setTestPreCopyToCallback(job->testPreCopyToCallback());
        }
        if (!job->standardResourceDirectory().isEmpty()) {
            entry->worker->setStandardResourceDirectory(job->standardResourceDirectory());
        }
        // Capture test worker result callback in entry so it persists after job destruction
        if (job->testWorkerResultCallback()) {
            entry->testWorkerResultCallback = job->testWorkerResultCallback();
        }
    }

    entry->worker->moveToThread(newThread);

    // Connections:
    // 1. Worker finished -> thread quit (DirectConnection, thread-safe)
    // 2. Worker finished -> manager.handleWorkerFinished (QueuedConnection, crosses to main thread)
    // 3. Thread finished -> worker.deleteLater (DirectConnection, runs during thread shutdown)
    // 4. Thread finished -> manager.reapEntry (QueuedConnection, crosses to main thread)

    QObject::connect(storedEntry->worker, &ArchiveExtractionWorker::finished,
                     newThread, &QThread::quit, Qt::DirectConnection);

    QObject::connect(storedEntry->worker, &ArchiveExtractionWorker::finished,
                     this, [this, handle](ArchiveExtractionWorker::ResultType result,
                                          const QStringList &files,
                                          const QString &errorMessage) {
        QMetaObject::invokeMethod(this, [this, handle, result, files, errorMessage]() {
            handleWorkerFinished(handle, result, files, errorMessage);
        }, Qt::QueuedConnection);
    }, Qt::DirectConnection);

    QObject::connect(newThread, &QThread::finished,
                     storedEntry->worker, &QObject::deleteLater, Qt::DirectConnection);

    QObject::connect(newThread, &QThread::finished,
                     this, [this, handle]() {
        QMetaObject::invokeMethod(this, [this, handle]() {
            ExtractionEntry *entry = findEntry(handle);
            if (entry) {
                reapEntry(entry);
            }
        }, Qt::QueuedConnection);
    }, Qt::DirectConnection);

    storedEntry->state.store(ExtractionEntry::State::Running, std::memory_order_release);
    QMetaObject::invokeMethod(storedEntry->worker, "start", Qt::QueuedConnection);

    return handle;
}

void ArchiveExtractionManager::requestCancellation(int handle)
{
    // Can be called from any thread (e.g., job destructor, doKill())
    // Step 1: Set the cancellation flag immediately (thread-safe atomic)
    std::shared_ptr<ExtractionEntry> entry;
    {
        QMutexLocker locker(&m_entriesMutex);
        if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
            entry = m_entries[handle];
        }
    }

    if (!entry) {
        return; // Entry already reaped
    }

    entry->cancellation->store(true, std::memory_order_release);

    // Step 2: Marshal thread-specific operation to manager thread via queued invocation.
    QMetaObject::invokeMethod(this, [this, handle]() {
        Q_ASSERT(QThread::currentThread() == thread());

        std::shared_ptr<ExtractionEntry> e;
        {
            QMutexLocker locker(&m_entriesMutex);
            if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
                e = m_entries[handle];
            }
        }

        if (!e) {
            return; // Entry already reaped - nothing to do
        }

        if (!e->testSuppressInterruptionFallback) {
            QThread *workerThread = e->thread;
            if (workerThread && workerThread->isRunning()) {
                workerThread->requestInterruption();
            }
        }
    }, Qt::QueuedConnection);
}

void ArchiveExtractionManager::handleWorkerFinished(int handle,
                                                    ArchiveExtractionWorker::ResultType result,
                                                    const QStringList &files,
                                                    const QString &errorMessage)
{
    // Must be called from manager's thread (main thread)
    Q_ASSERT(QThread::currentThread() == thread());

    std::shared_ptr<ExtractionEntry> entry;
    {
        QMutexLocker locker(&m_entriesMutex);
        if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
            entry = m_entries[handle];
        }
    }

    if (!entry) {
        return; // Already reaped/finalized
    }

    // Reentrancy guard: duplicate or reentrant result notifications are harmless.
    if (entry->resultClaimed.exchange(true, std::memory_order_acq_rel)) {
        return;
    }

    // Capture QPointer before dispatching; never dereference after a handler may emit result.
    QPointer<ArchiveExtractionJob> job = entry->job;

    // The worker's emitted result is authoritative. Do NOT override Success/FallbackRequired
    // with Cancelled based on a late-arriving cancellation flag.
    if (job) {
        if (entry->job->testWorkerResultCallback()) {
            entry->job->testWorkerResultCallback()(result);
        }
        switch (result) {
            case ArchiveExtractionWorker::Success:
                entry->job->handleSuccess(files);
                break;
            case ArchiveExtractionWorker::FallbackRequired:
                entry->job->handleFallbackRequired();
                break;
            case ArchiveExtractionWorker::ExtractionFailed:
                entry->job->handleExtractionFailed(errorMessage);
                break;
            case ArchiveExtractionWorker::Cancelled:
                entry->job->handleCancelled();
                break;
        }
    } else {
        if (entry->testWorkerResultCallback) {
            entry->testWorkerResultCallback(result);
        }
    }

    // Result disposition is complete from the manager's perspective once handlers ran.
    entry->resultDisposed.store(true, std::memory_order_release);

    if (!job) {
        // Job object already destroyed before result delivery. No live consumer remains.
        entry->jobDead.store(true, std::memory_order_release);
        // If the thread is already reaped, clean up any unconsumed payload now.
        // Otherwise reapEntry will handle it once threadReaped becomes true.
        if (entry->threadReaped.load(std::memory_order_acquire)) {
            cleanupPayloadIfNeeded(entry);
        }
    }

    // Attempt direct reaping if the thread has already finished.
    QThread *workerThread = entry->thread;
    if (workerThread && !entry->threadReaped.load(std::memory_order_acquire)) {
        if (!workerThread->isRunning() && workerThread->wait(0)) {
            delete workerThread;
            entry->thread = nullptr;
            entry->worker = nullptr;
            entry->threadReaped.store(true, std::memory_order_release);
            // If the job is dead, clean up any unconsumed payload now that
            // both result delivery and physical reaping are complete.
            if (entry->jobDead.load(std::memory_order_acquire)) {
                cleanupPayloadIfNeeded(entry);
            }
        } else if (workerThread->isRunning()) {
            entry->state.store(ExtractionEntry::State::Finishing, std::memory_order_release);
            scheduleReapRetry(handle);
            return;
        }
    }

    // Physical reaping happens independently via reapEntry (scheduled by thread.finished).
    // Finalize only when both resultDisposed && threadReaped are true.
    tryFinalize(entry);
}

void ArchiveExtractionManager::deliverSyntheticFailure(std::shared_ptr<ExtractionEntry> entry,
                                                       const QString &errorMessage)
{
    // Synthetic terminal failure for a thread that failed to start.
    // No worker was ever created/moved, so no worker result exists; report failure directly.
    QPointer<ArchiveExtractionJob> job = entry->job;
    entry->resultClaimed.store(true, std::memory_order_release);

    if (job) {
        entry->job->handleExtractionFailed(errorMessage);
    }

    entry->resultDisposed.store(true, std::memory_order_release);
    if (entry->jobDead.load(std::memory_order_acquire) || entry->job.isNull()) {
        cleanupPayloadIfNeeded(entry);
    }
    tryFinalize(entry);
}

void ArchiveExtractionManager::cleanupPayloadIfNeeded(std::shared_ptr<ExtractionEntry> entry)
{
    // Remove an unconsumed temp payload only when both notifications are complete:
    // resultDisposed (job notification/payload disposition complete) AND
    // threadReaped (worker finished using the payload). The caller is responsible
    // for ensuring no live job remains to consume the payload (checked via jobDead
    // flag or QPointer::isNull() before calling this function).
    // QFileInfo::exists guards against double-removal when a live fallback
    // handler already renamed the file.
    if (!entry->resultDisposed.load(std::memory_order_acquire)
        || !entry->threadReaped.load(std::memory_order_acquire)) {
        return;
    }

    const QString &payload = entry->payloadFile;
    if (!payload.isEmpty() && QFileInfo::exists(payload)) {
        if (!QFile::remove(payload)) {
            qCWarning(KNEWSTUFFCORE) << "Failed to remove leftover payload" << payload
                                     << "for handle" << entry->handle
                                     << QFile().errorString();
        }
    }
}

void ArchiveExtractionManager::reapEntry(ExtractionEntry *entry)
{
    // Must be called from manager's thread (main thread)
    Q_ASSERT(QThread::currentThread() == thread());

    if (!entry) {
        return;
    }

    auto currentState = entry->state.load(std::memory_order_acquire);
    if (currentState == ExtractionEntry::State::Done) {
        return; // Already finalized
    }
    if (currentState == ExtractionEntry::State::Reaping) {
        return; // Reaping in progress, avoid reentrancy
    }

    entry->state.store(ExtractionEntry::State::Reaping, std::memory_order_release);

    // Obtain a shared_ptr that shares ownership with the vector's entry,
    // so the entry cannot be destroyed while we are working on it.
    std::shared_ptr<ExtractionEntry> sharedEntry;
    {
        QMutexLocker locker(&m_entriesMutex);
        if (entry->handle >= 0 && entry->handle < static_cast<int>(m_entries.size())) {
            sharedEntry = m_entries[entry->handle];
        }
    }
    if (!sharedEntry) {
        return; // Entry already finalized
    }

    QThread *workerThread = sharedEntry->thread;

    if (!workerThread) {
        // No thread (startup failure or test mode); physical side already complete.
        sharedEntry->threadReaped.store(true, std::memory_order_release);
        tryFinalize(sharedEntry);
        return;
    }

    if (workerThread->isRunning()) {
        // Thread has not yet stopped. Reset state so a later retry can proceed;
        // do not call quit() as a substitute for joining; the worker's
        // finished->thread.quit connection remains responsible for stopping.
        // Schedule a retry; do not block the GUI thread.
        sharedEntry->state.store(ExtractionEntry::State::Finishing, std::memory_order_release);
        scheduleReapRetry(sharedEntry->handle);
        return;
    }

    // Thread has finished. Attempt non-blocking join.
    if (!workerThread->wait(0)) {
        // join/cleanup still ongoing (e.g. TLS destructors). Reset state so a
        // later retry can proceed, then schedule a retry.
        sharedEntry->state.store(ExtractionEntry::State::Finishing, std::memory_order_release);
        scheduleReapRetry(sharedEntry->handle);
        return;
    }

    // wait(0) returned true for a thread known to have started successfully: safe to delete.
    delete workerThread;
    sharedEntry->thread = nullptr;
    sharedEntry->worker = nullptr; // already deleteLater'd via thread.finished connection

    sharedEntry->threadReaped.store(true, std::memory_order_release);

    // tryFinalize defers removal if the job is still alive (waits for destroyed signal).
    tryFinalize(sharedEntry);
}

void ArchiveExtractionManager::scheduleReapRetry(int handle)
{
    if (handle < 0) {
        return;
    }

    // Ensure only one retry is outstanding per entry.
    std::shared_ptr<ExtractionEntry> e;
    {
        QMutexLocker locker(&m_entriesMutex);
        if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
            e = m_entries[handle];
        }
    }
    if (!e) {
        return;
    }
    if (e->reapRetryScheduled) {
        return;
    }
    e->reapRetryScheduled = true;

    // Context-bound single-shot timer on the manager; modest non-zero interval.
    // 15 ms gives the extraction thread time to complete TLS cleanup without busy-spinning.
    QTimer::singleShot(15, this, [this, handle]() {
        onReapTimer(handle);
    });
}

void ArchiveExtractionManager::onReapTimer(int handle)
{
    Q_ASSERT(QThread::currentThread() == thread());

    std::shared_ptr<ExtractionEntry> e;
    {
        QMutexLocker locker(&m_entriesMutex);
        if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
            e = m_entries[handle];
        }
    }

    if (!e) {
        return; // Entry already finalized; stale callback is a no-op.
    }

    e->reapRetryScheduled = false;
    reapEntry(e.get());
}

bool ArchiveExtractionManager::tryFinalize(std::shared_ptr<ExtractionEntry> entry)
{
    if (!entry) {
        return false;
    }

    // Finalization invariant: both result delivery and physical reaping must be complete.
    if (!entry->resultDisposed.load(std::memory_order_acquire)
        || !entry->threadReaped.load(std::memory_order_acquire)) {
        return false;
    }

    if (entry->state.load(std::memory_order_acquire) == ExtractionEntry::State::Done) {
        // Already marked Done. Clean up payload if job is dead, then release entry.
        if (entry->jobDead.load(std::memory_order_acquire)
            || entry->job.isNull()) {
            cleanupPayloadIfNeeded(entry);
        }
        releaseEntryFromVector(entry);
        return true;
    }

    // Clean up any unconsumed payload only if the job is dead (no live consumer).
    // A live fallback handler may still consume/rename the payload; the manager
    // must not remove it before that handler has returned.
    if (entry->jobDead.load(std::memory_order_acquire)
        || entry->job.isNull()) {
        cleanupPayloadIfNeeded(entry);
    }

    entry->state.store(ExtractionEntry::State::Done, std::memory_order_release);

    // Do NOT remove from vector yet if the job is still alive — the job's
    // destroyed signal callback will call tryFinalize again (Done branch),
    // which will set jobDead, clean up payload, and release the entry.
    // If the job is already dead/null, release immediately.
    if (entry->jobDead.load(std::memory_order_acquire)
        || entry->job.isNull()) {
        releaseEntryFromVector(entry);
    }

    return true;
}

bool ArchiveExtractionManager::waitForAllFinished(int timeoutMs)
{
    Q_ASSERT(QThread::currentThread() == thread());

    QElapsedTimer timer;
    timer.start();

    while (true) {
        int active = 0;
        {
            QMutexLocker locker(&m_entriesMutex);
            for (const auto &entry : m_entries) {
                if (entry && entry->state.load(std::memory_order_acquire) != ExtractionEntry::State::Done) {
                    active++;
                }
            }
        }

        if (active == 0) {
            return true;
        }

        if (timer.elapsed() >= timeoutMs) {
            return false;
        }

        QCoreApplication::processEvents(QEventLoop::AllEvents, 50);
    }
}

int ArchiveExtractionManager::activeExtractionsCount() const
{
    QMutexLocker locker(&m_entriesMutex);
    int count = 0;
    for (const auto &entry : m_entries) {
        if (entry && entry->state.load(std::memory_order_acquire) != ExtractionEntry::State::Done) {
            count++;
        }
    }
    return count;
}

ArchiveExtractionManager::ExtractionEntry* ArchiveExtractionManager::findEntry(int handle)
{
    QMutexLocker locker(&m_entriesMutex);
    if (handle >= 0 && handle < static_cast<int>(m_entries.size())) {
        return m_entries[handle].get();
    }
    return nullptr;
}

    if (m_cancellation && m_cancellation->load(std::memory_order_acquire)) {

void ArchiveExtractionWorker::doWork()
{
    QStringList installedFiles;
    ResultType result = Success;
    QString errorMessage;
    bool isArchive = true;

    // Test seam: allow test to block after started() is emitted
    Q_EMIT started();
    if (m_testStartedCallback) {
        m_testStartedCallback();
    }
    if (m_testBarrierAfterStarted) {
        m_testBarrierAfterStarted->acquire();
    }

    qCDebug(KNEWSTUFFCORE) << "Worker doWork started for" << d->payloadFile;

    // Check for cancellation before starting
    if (isCancellationRequested()) {
        QFile::remove(d->payloadFile); // Cancelled: no fallback, clean up payload.
        Q_EMIT finished(Cancelled, QStringList(), QString());
        return;
    }

    ArchiveExtractionWorker::UncompressionOption uncompressionOpt = d->uncompressionOption;

    // Respect the uncompress flag
    if (uncompressionOpt == AlwaysUncompress || uncompressionOpt == UncompressIntoSubdirIfArchive
        || uncompressionOpt == UncompressIfArchive || uncompressionOpt == UncompressIntoSubdir) {

        // Check MIME type to determine if it's an archive
        QMimeDatabase db;
        QMimeType mimeType = db.mimeTypeForFile(d->payloadFile);

        QScopedPointer<KArchive> archive;

        if (mimeType.inherits(QStringLiteral("application/zip"))) {
            archive.reset(new KZip(d->payloadFile));
        } else if (mimeType.inherits(QStringLiteral("application/tar"))
                || mimeType.inherits(QStringLiteral("application/x-tar"))
                || mimeType.inherits(QStringLiteral("application/x-gzip"))
                || mimeType.inherits(QStringLiteral("application/x-bzip"))
                || mimeType.inherits(QStringLiteral("application/x-lzma"))
                || mimeType.inherits(QStringLiteral("application/x-xz"))
                || mimeType.inherits(QStringLiteral("application/x-bzip-compressed-tar"))
                || mimeType.inherits(QStringLiteral("application/x-compressed-tar"))) {
            archive.reset(new KTar(d->payloadFile));
        } else {
            // Unknown MIME type
            if (uncompressionOpt == AlwaysUncompress || uncompressionOpt == UncompressIntoSubdir) {
                // Wallpaper fallback: non-archive wallpapers should use FallbackRequired
                if (m_standardResourceDirectory == QLatin1String("wallpaper")) {
                    result = FallbackRequired;
                    Q_EMIT finished(result, QStringList(), QString());
                    return;
                }
                // Not a wallpaper: treat as extraction failure
                errorMessage = i18n("Could not determine the type of archive of the downloaded file %1", d->payloadFile);
                result = ExtractionFailed;
                QFile::remove(d->payloadFile); // No consumer for verified paths; clean up.
                Q_EMIT finished(result, QStringList(), errorMessage);
                return;
            }
            // For UncompressIfArchive, UncompressIntoSubdirIfArchive:
            // treat as non-archive (will emit FallbackRequired)
            isArchive = false;
        }

        if (isArchive) {
            // Check for cancellation before opening archive
            if (isCancellationRequested()) {
                QFile::remove(d->payloadFile); // Cancelled: no fallback, clean up payload.
                Q_EMIT finished(Cancelled, QStringList(), QString());
                return;
            }

            bool success = archive->open(QIODevice::ReadOnly);
            if (!success) {
                // Archive open failure - recognized archive type but corrupt/unreadable
                // Wallpaper fallback does NOT apply here - only for non-archive files (unknown MIME type)
                if (uncompressionOpt == AlwaysUncompress) {
                    errorMessage = i18n("Failed to open the archive file %1. The reported error was: %2", d->payloadFile, archive->errorString());
                    result = ExtractionFailed;
                    QFile::remove(d->payloadFile); // No consumer for verified paths; clean up.
                    Q_EMIT finished(result, QStringList(), errorMessage);
                    return;
                }
                // For UncompressIfArchive and UncompressIntoSubdirIfArchive: fall back to non-archive handling
                if (uncompressionOpt == UncompressIfArchive || uncompressionOpt == UncompressIntoSubdirIfArchive) {
                    // Cancellation checkpoint before emitting FallbackRequired
                    if (isCancellationRequested()) {
                        archive->close();
                        QFile::remove(d->payloadFile); // Cancelled: no fallback, clean up payload.
                        Q_EMIT finished(Cancelled, QStringList(), QString());
                        return;
                    }
                    result = FallbackRequired;
                    Q_EMIT finished(result, QStringList(), QString());
                    return;
                }
                // UncompressIntoSubdir: archive open failure is an extraction failure
                errorMessage = i18n("Failed to open the archive file %1. The reported error was: %2", d->payloadFile, archive->errorString());
                result = ExtractionFailed;
                QFile::remove(d->payloadFile); // No consumer for verified paths; clean up.
                Q_EMIT finished(result, QStringList(), errorMessage);
                return;
            }
        }

        if (isArchive) {
            const KArchiveDirectory *dir = archive->directory();
            if (!dir) {
                // Defensive: archive opened but has no root directory
                errorMessage = i18n("Archive %1 has no valid root directory", d->payloadFile);
                result = ExtractionFailed;
                archive->close();
                QFile::remove(d->payloadFile);
                Q_EMIT finished(result, QStringList(), errorMessage);
                return;
            }
            // If there is more than an item in the file, and we are requested to do so
            // put contents in a subdirectory with the same name as the file
            QString installPath;
            const bool isSubdir =
                (uncompressionOpt == UncompressIntoSubdir || uncompressionOpt == UncompressIntoSubdirIfArchive) && dir->entries().count() > 1;
            if (isSubdir) {
                installPath = d->installDir + QLatin1Char('/') + QFileInfo(d->payloadFile).baseName();
            } else {
                installPath = d->installDir;
            }

            // Protect destination with per-destination mutex
            QMutex* destMutex = KNSCore::getDestinationMutex(installPath);
            qCDebug(KNEWSTUFFCORE) << "Acquiring mutex for" << installPath;
            QMutexLocker locker(destMutex);
            qCDebug(KNEWSTUFFCORE) << "Mutex acquired for" << installPath;

            // Test seam: allow test to block after mutex acquisition but before cancellation check
            if (m_testBeforeCancellationCheckCallback) {
                m_testBeforeCancellationCheckCallback();
            }
            if (m_testBarrierBeforeCancellationCheck) {
                m_testBarrierBeforeCancellationCheck->acquire();
            }

            if (isCancellationRequested()) {
                archive->close(); // Archive was opened, close it before removing payload
                QFile::remove(d->payloadFile); // Cancelled: no fallback, clean up payload.
                Q_EMIT finished(Cancelled, QStringList(), QString());
                return;
            }

            // Test seam: allow test to block before copyTo
            Q_EMIT aboutToCopyTo();
            if (m_testAboutToCopyToCallback) {
                m_testAboutToCopyToCallback();
            }
            if (m_testBeforeCopyToCallback) {
                m_testBeforeCopyToCallback();
            }
            if (m_testBarrierBeforeCopyTo) {
                m_testBarrierBeforeCopyTo->acquire();
            }

            // Cancellation checkpoint after barrierBeforeCopyTo and before copyTo().
            // This allows tests to verify cancellation can prevent copyTo() from starting.
            // Note: if cancellation arrives while worker is in copyTo(), it will only be
            // observed after copyTo() returns (see checkpoint below).
            if (isCancellationRequested()) {
                archive->close(); // Archive was opened, close it before removing payload
                QFile::remove(d->payloadFile); // Cancelled: no fallback, clean up payload.
                Q_EMIT finished(Cancelled, QStringList(), QString());
                return;
            }

            // Test-only observation point: fires after the pre-copyTo cancellation checkpoint
            // but before the actual copyTo() call. Proves the checkpoint was passed.
            if (m_testPreCopyToCallback) {
                m_testPreCopyToCallback();
            }

            // copyTo() is not interruptible; if cancellation is requested during extraction,
            // it will only take effect after copyTo() completes. The worker checks for
            // interruption after copyTo() returns and will then clean up and emit Cancelled.
            // The destination directory may be partially populated; no rollback is performed.
            qCDebug(KNEWSTUFFCORE) << "Starting copyTo to" << installPath;
            bool copySuccess = dir->copyTo(installPath);
            qCDebug(KNEWSTUFFCORE) << "copyTo finished, success=" << copySuccess;

            if (isCancellationRequested()) {
                archive->close(); // Archive was opened, close it before removing payload
                QFile::remove(d->payloadFile); // Cancelled: no fallback, clean up payload.
                Q_EMIT finished(Cancelled, QStringList(), QString());
                return;
            }

            if (copySuccess) {
                // If we extracted the subdir we want to save it using the /* notation
                if (isSubdir) {
                    installedFiles << QDir(installPath).absolutePath() + QLatin1String("/*");
                } else {
                    // Collect all installed files
                    const auto lst = dir->entries();
                    for (const QString &entry : lst) {
                        const auto currentEntry = dir->entry(entry);
                        const QString childPath = QDir(installPath).filePath(entry);
                        if (currentEntry->isFile()) {
                            installedFiles << childPath;
                        } else if (currentEntry->isDirectory()) {
                            installedFiles << childPath + QStringLiteral("/*");
                        }
                    }
                }
                // If copyTo succeeded but no files were extracted, treat as extraction failure
                // (corrupt/empty archive) rather than success with empty list.
                if (installedFiles.isEmpty()) {
                    // Empty/corrupt archive - wallpaper fallback does NOT apply
                    errorMessage = i18n("Archive %1 is empty or corrupt", d->payloadFile);
                    result = ExtractionFailed;
                    archive->close();
                    QFile::remove(d->payloadFile);
                    Q_EMIT finished(result, QStringList(), errorMessage);
                    return;
                }
            } else {
                // copyTo() failure: no installed files can be produced, and no caller can retry
                // or consume the payload (installNonArchiveFile is not invoked for ExtractionFailed).
                // Treat as extraction failure and clean up the payload.
                qCWarning(KNEWSTUFFCORE) << "could not install" << d->payloadFile << "to" << installPath;
                errorMessage = i18n("Failed to extract archive %1 to %2", d->payloadFile, installPath);
                result = ExtractionFailed;
                archive->close();
                QFile::remove(d->payloadFile);
                Q_EMIT finished(result, QStringList(), errorMessage);
                return;
            }

            archive->close();
            QFile::remove(d->payloadFile);
        }
    }

    // If not an archive, determine the result type
    if (!isArchive) {
        // For UncompressIntoSubdir with unknown MIME or archive open failure: empty list = failure
        // For UncompressIfArchive/UncompressIntoSubdirIfArchive: fallback required
        if (uncompressionOpt == UncompressIfArchive || uncompressionOpt == UncompressIntoSubdirIfArchive) {
            // Cancellation checkpoint before emitting FallbackRequired
            if (isCancellationRequested()) {
                QFile::remove(d->payloadFile); // Cancelled: no fallback, clean up payload.
                Q_EMIT finished(Cancelled, QStringList(), QString());
                return;
            }
            result = FallbackRequired;
        } else {
            // UncompressIntoSubdir, AlwaysUncompress (already handled above), NeverUncompress
            result = Success; // empty installedFiles will be treated as failure by caller
        }
    }

    qCDebug(KNEWSTUFFCORE) << "Worker doWork finished, emitting finished with result" << result;
    Q_EMIT finished(result, installedFiles, errorMessage);
}

{
    QMutexLocker locker(&s_mutexMapLock);
    // Normalize the path to a directory:
    // - For existing paths: use canonical path of the directory (for dirs) or parent directory (for files)
    // - For non-existing paths: use absolute path of the directory (for dirs) or parent directory (for files)
    QFileInfo fi(path);
    QString canonical;
    if (fi.exists()) {
        if (fi.isDir()) {
            canonical = fi.canonicalFilePath();
        } else {
            canonical = fi.canonicalPath();
        }
    } else {
        // Path doesn't exist - determine if it's a file or directory by checking if the parent exists
        QFileInfo parentFi(fi.path());
        if (parentFi.exists() && parentFi.isDir()) {
            // Parent directory exists, use its canonical path
            canonical = parentFi.canonicalFilePath();
        } else {
            // Fall back to absolute path of parent directory
            canonical = fi.absolutePath();
        }
    }
    if (canonical.isEmpty()) {
        canonical = fi.absolutePath();
    }
    QMutex* mutex = s_destinationMutexes.value(canonical);
    if (!mutex) {
        mutex = new QMutex();
        s_destinationMutexes.insert(canonical, mutex);
    }
    return mutex;
}

void clearDestinationMutexes()
{
    QMutexLocker locker(&s_mutexMapLock);
    for (QMutex* mutex : s_destinationMutexes) {
        delete mutex;
    }
    s_destinationMutexes.clear();
}

using SemaphorePtr = std::shared_ptr<QSemaphore>;

// Thread-safe test state holder for callbacks that may execute asynchronously
// Uses shared_ptr + atomic to ensure lifetime extends beyond test scope
template<typename T>
struct TestState {
    std::shared_ptr<std::atomic<T>> value;
    TestState(T initial = T{}) : value(std::make_shared<std::atomic<T>>(initial)) {}
    T load() const { return value->load(std::memory_order_acquire); }

    void increment() { value->fetch_add(1, std::memory_order_acq_rel); }

// Robust scope guard for test cleanup
// Ensures cancellation is requested, barriers are released, and worker/manager completion waited for
// CRITICAL: Bounded wait with safe termination strategy.
// - Requests cancellation FIRST if job is still active (so workers observe cancellation at next checkpoint)
// - Releases barriers AFTER cancellation requested (unblocks workers to hit their cancellation checkpoints)
// - Waits for manager cleanup with generous timeout
// - If timeout expires with active workers, calls std::quick_exit to avoid unsafe process exit
// - Never destroys a running QThread. Never frees worker-owned state while its thread may still be using it.
// - Never uses nested GUI event loops or unsafe forced deletion as a workaround.
// - Note: KArchiveDirectory::copyTo() is non-interruptible; a worker blocked in copyTo() will not see
//   cancellation until copyTo() returns. This is an inherent limitation of the archive library.
struct TestScope {
    explicit TestScope(ArchiveExtractionJob* job = nullptr)
        : m_job(job)
        , m_manager(&ArchiveExtractionManager::instance())
    {}

    // Non-copyable, non-movable: owns barriers and manages cleanup; copying would duplicate
    // barrier ownership and cause double-release or missed cleanup.
    TestScope(const TestScope&) = delete;
    TestScope& operator=(const TestScope&) = delete;
    TestScope(TestScope&&) = delete;
    TestScope& operator=(TestScope&&) = delete;

    ~TestScope() {
        // 1. Request cancellation FIRST if job still exists and hasn't finished.
        // This ensures that when barriers are released, workers will observe
        // the cancellation flag at their next checkpoint.
        if (m_job && m_job->error() == KJob::NoError) {
            m_job->kill();
        }

        // 2. Release all barriers to unblock any waiting worker at their checkpoints.
        // Ordering: cancellation requested FIRST, then barriers released.
        // Workers resume and immediately hit their next cancellation checkpoint.
        for (auto& s : m_barriers) {
            if (s) s->release();
        }

        // 3. Wait for manager to clean up all entries with bounded timeout
        constexpr int cleanupTimeoutMs = 30000; // Generous timeout for worker completion + thread shutdown
        QElapsedTimer timer;
        timer.start();
        while (!m_manager->waitForAllFinished(1000)) {
            QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
            if (timer.elapsed() >= cleanupTimeoutMs) {
                // Timeout: emit diagnostic with manager/entry state
                qCritical() << "TestScope cleanup timeout after" << cleanupTimeoutMs << "ms";
                qCritical() << "Active extractions count:" << m_manager->activeExtractionsCount();
                qCritical() << "Worker threads may still be running. Terminating test process to avoid unsafe exit.";
                
                // Use std::quick_exit to terminate without destroying C++ objects
                // (which could race with still-running worker threads).
                // This is the safest option when threads cannot be joined.
                std::quick_exit(EXIT_FAILURE);
            }
        }

        // 4. Process any remaining events
        QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
    }

    // Register a barrier that should be released on cleanup
    void addBarrier(SemaphorePtr barrier) {
        if (barrier) m_barriers.push_back(std::move(barrier));
    }

    // Register the job under test
    void setJob(ArchiveExtractionJob* job) {
        m_job = job;
    }

private:
    QPointer<ArchiveExtractionJob> m_job;
    ArchiveExtractionManager* m_manager = nullptr;
    std::vector<SemaphorePtr> m_barriers;
};

// Helper to create a shared semaphore
inline SemaphorePtr makeSemaphore(int initial = 0) {
    return std::make_shared<QSemaphore>(initial);
}

// Safe wrapper for waiting on a semaphore with timeout
// Returns true if acquired, false on timeout
inline bool tryAcquire(SemaphorePtr sem, int timeoutMs = 5000) {
    return sem && sem->tryAcquire(1, timeoutMs);
}

// Helper to release a shared semaphore
inline void release(SemaphorePtr sem) {

}

    // If this fires, the checkpoint was passed (cancellation not observed at checkpoint).
    // If cancellation is requested while worker is blocked at barrier, the checkpoint should

    QVERIFY(tryAcquire(workerAtBarrierSem, 5000));

    // Create jobs with both shared and distinct destinations
    for (int i = 0; i < numJobs; ++i) {
        tempArchives.emplace_back();
        QVERIFY(tempArchives[i].open());
        QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
        QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
        tempArchives[i]->write(sourceArchive.readAll());
        tempArchives[i]->flush();
        sourceArchive.close();
        QString payloadFile = tempArchives[i].fileName();
        payloadFiles.push_back(payloadFile);

        // Alternate between same destination and distinct destinations
        QString jobInstallDir = installDir;
        if (i % 2 == 1) {
            jobInstallDir = installDir + QStringLiteral("subdir%1/").arg(i);
            QDir().mkpath(jobInstallDir);
        }

 ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, jobInstallDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive, &jobParent);
         job->setAutoDelete(false);
         auto spy = std::make_unique<QSignalSpy>(job, &KJob::result);
         jobs.push_back(job);
         spies.push_back(std::move(spy));

        // Set up scope guard for each job
        scopes.emplace_back(std::make_unique<TestScope>(job));
    }

    // Start all jobs
    for (int i = 0; i < numJobs; ++i) {
        jobs[i]->start();
    }

    // Collect handles
    for (int i = 0; i < numJobs; ++i) {
        int handle = jobs[i]->extractionHandle();
        QVERIFY(handle != -1);
        handles.push_back(handle);
    }

    // Verify all entries are active
    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), numJobs);

    // Wait for all jobs to complete - use collective event-driven wait to avoid
    // race where a job finishes before its individual wait() begins.
    QElapsedTimer waitTimer;
    waitTimer.start();
    const int waitTimeout = 30000;
    while (true) {
        bool allDone = true;
        for (int i = 0; i < numJobs; ++i) {
            if (spies[i]->count() < 1) {
                allDone = false;
                break;
            }
        }
        if (allDone) {
            break;
        }
        if (waitTimer.elapsed() >= waitTimeout) {
            // Report which jobs didn't complete
            for (int i = 0; i < numJobs; ++i) {
                if (spies[i]->count() < 1) {
                    QFAIL(qPrintable(QString("Job %1 did not complete within timeout").arg(i)));
                }
            }
            break;
        }
        QCoreApplication::processEvents(QEventLoop::AllEvents, 50);
    }

    // Verify all jobs completed exactly once
    for (int i = 0; i < numJobs; ++i) {
        QCOMPARE(spies[i]->count(), 1);
        QVERIFY(!jobs[i]->error());
    }

    // Manager entry count should return to zero
    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(30000));
    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 0);

    // All payloads should be cleaned up
    for (const QString &pf : payloadFiles) {
        QVERIFY(!QFileInfo::exists(pf));
    }

    // Scopes automatically handle cleanup - jobs are deleted via jobParent,
    // spies are auto-deleted via unique_ptr, tempArchives destroyed after jobParent
}

void ArchiveExtractionJobTest::testJobDestructionBeforeResultDelivery()
{
    // C1/C4: Job destroyed before the worker produces a result. The manager
    // must not crash or access freed memory, and the entry must be released.
    auto barrier = makeSemaphore();
    auto workerStartedSem = makeSemaphore();
    ScopedTempFile tempArchive;
    QVERIFY(tempArchive.open());
    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
    tempArchive->write(sourceArchive.readAll());
    tempArchive->flush();
    sourceArchive.close();
    QString payloadFile = tempArchive.fileName();

    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation)
                         + QStringLiteral("/demo_testJobDestructionBeforeResult/");
    QDir().mkpath(installDir);

    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir,
                                                         ArchiveExtractionJob::UncompressIntoSubdirIfArchive);
    job->setTestBarrierAfterStarted(barrier);
    job->setTestWorkerStartedCallback([workerStartedSem]() { workerStartedSem->release(); });

    QSignalSpy spy(job, &KJob::result);
    job->start();

    QVERIFY(tryAcquire(workerStartedSem, 5000));
    int handle = job->extractionHandle();
    QVERIFY(handle != -1);
    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 1);

    // Destroy the job while the worker is blocked at the barrier (before result).
    job->deleteLater();
    QPointer<ArchiveExtractionJob> jobPtr(job);
    QTRY_VERIFY(jobPtr.isNull());

    // Release the barrier so the worker can finish (it will see cancellation).
    release(barrier);

    // Manager should release the entry once the thread finishes.
    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(10000));
    QCOMPARE(ArchiveExtractionManager::instance().testTotalEntryCount(), 0);
}

void ArchiveExtractionJobTest::testSyntheticFailureJobDestroyedByHandler()
{
    // C4: Prove that the synthetic-failure path cannot use the job after a
    // synchronous result handler destroys it. The job must have a valid
    // handle/state before the result callback runs.
    ArchiveExtractionManager::instance().setTestForceStartFailure(true);

    bool handlerDestroyedJob = false;
    bool resultReceived = false;

    ScopedTempFile tempArchive;
    QVERIFY(tempArchive.open());
    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
    tempArchive->write(sourceArchive.readAll());
    tempArchive->flush();
    sourceArchive.close();
    QString payloadFile = tempArchive.fileName();

    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation)
                         + QStringLiteral("/demo_testSyntheticFailure/");
    QDir().mkpath(installDir);

    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir,
                                                         ArchiveExtractionJob::UncompressIntoSubdirIfArchive);
    job->setAutoDelete(false);

    // Connect a result handler that destroys the job synchronously.
    QObject::connect(job, &KJob::result, job, [&handlerDestroyedJob, &resultReceived, job](KJob *finishedJob) {
        resultReceived = true;
        QVERIFY(finishedJob == job);  // Same job object
        QVERIFY(job->extractionHandle() != -1);  // Handle must be valid before result
        handlerDestroyedJob = true;
        finishedJob->deleteLater();
    });

    job->start();

    QTRY_VERIFY_WITH_TIMEOUT(resultReceived, 5000);
    QVERIFY(handlerDestroyedJob);
    QVERIFY(job->error());
    QCOMPARE(job->error(), KJob::UserDefinedError);

    // Entry should be released (job was destroyed, then tryFinalize removed it).
    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(5000));
    QCOMPARE(ArchiveExtractionManager::instance().testTotalEntryCount(), 0);

    // Payload removed by implementation (no worker was created).
    QVERIFY(!QFileInfo::exists(payloadFile));

    // Reset the seam.
    ArchiveExtractionManager::instance().setTestForceStartFailure(false);

    QPointer<ArchiveExtractionJob> jobPtr(job);
    QTRY_VERIFY(jobPtr.isNull());
}
