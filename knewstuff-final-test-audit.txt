    32	// Shared ownership for test synchronization objects
    33	// Ensures barriers and callbacks remain alive as long as any worker could access them
    34	using SemaphorePtr = std::shared_ptr<QSemaphore>;
    35	
    36	// Robust scope guard for test cleanup
    37	// Ensures barriers are released, cancellation requested, and worker/manager completion waited for
    38	// CRITICAL: Does NOT time out. Waits indefinitely (with event processing) to ensure
    39	// all workers complete before test resources (payload files, jobs, semaphores) are destroyed.
    40	// TestScope is destroyed first (declared last), so all test resources are still alive.
    41	// If a worker is truly stuck, the test will hang - this is correct behavior for a stuck test.
    42	struct TestScope {
    43	    explicit TestScope(ArchiveExtractionJob* job = nullptr)
    44	        : m_job(job)
    45	        , m_manager(&ArchiveExtractionManager::instance())
    46	    {}
    47	
    48	    ~TestScope() {
    49	        // 1. Release all barriers to unblock any waiting worker
    50	        for (auto& s : m_barriers) {
    51	            if (s) s->release();
    52	        }
    53	
    54	        // 2. Request cancellation if job still exists
    55	        // We check if job has no error (i.e., hasn't finished with an error or success)
    56	        if (m_job && m_job->error() == KJob::NoError) {
    57	            m_job->kill();
    58	        }
    59	
    60	        // 3. Wait for manager to clean up all entries
    61	        // CRITICAL: NO TIMEOUT. Wait indefinitely (with event processing) to ensure
    62	        // all workers complete before test resources (payload files, jobs, semaphores) are destroyed.
    63	        // TestScope is destroyed first (declared last), so all test resources are still alive.
    64	        // If a worker is truly stuck, the test will hang - this is correct behavior for a stuck test.
    65	        while (!m_manager->waitForAllFinished(1000)) {
    66	            QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
    67	        }
    68	
    69	        // 4. Process any remaining events
    70	        QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
    71	    }
    72	
    73	    // Register a barrier that should be released on cleanup
    74	    void addBarrier(SemaphorePtr barrier) {
    75	        if (barrier) m_barriers.push_back(std::move(barrier));
    76	    }
    77	
    78	    // Register the job under test
    79	    void setJob(ArchiveExtractionJob* job) {
    80	        m_job = job;
    81	    }
    82	
    83	private:
    84	    QPointer<ArchiveExtractionJob> m_job;
    85	    ArchiveExtractionManager* m_manager = nullptr;
    86	    std::vector<SemaphorePtr> m_barriers;
    87	};

   194	void ArchiveExtractionJobTest::cleanup()
   195	{
   196	    // Wait for any pending extractions to complete before next test
   197	    // This ensures clean state between tests since manager is a singleton
   198	    const bool cleaned = ArchiveExtractionManager::instance().waitForAllFinished(5000);
   199	    if (!cleaned) {
   200	        qWarning() << "cleanup(): Manager did not finish all extractions within timeout";
   201	        QFAIL("Manager cleanup timed out between tests - dangling extractions may contaminate next test");
   202	    }
   203	
   204	    // Do not clear destination mutexes here - mutexes are process-lifetime and
   205	    // clearing them while worker threads might still be running is unsafe.
   206	    // Mutexes will be cleaned up at process exit.
   207	}

   485	void ArchiveExtractionJobTest::testCancellationAfterWorkerCreationBeforeExecution()
   486	{
   487	    // Test cancellation after worker thread created but before worker executes doWork()
   488	    // Uses barrier to deterministically block worker at startup checkpoint
   489	    auto barrier = makeSemaphore();
   490	    auto workerStartedSem = makeSemaphore();
   491	    ScopedTempFile tempArchive;
   492	    QVERIFY(tempArchive.open());
   493	    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
   494	    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
   495	    tempArchive->write(sourceArchive.readAll());
   496	    tempArchive->flush();
   497	    sourceArchive.close();
   498	    QString payloadFile = tempArchive.fileName();
   499	
   500	    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation) + QLatin1String("/demo/");
   501	    QDir().mkpath(installDir);
   502	
   503	    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive);
   504	
   505	    // Configure worker to block at the barrier after started() is emitted
   506	    job->setTestBarrierAfterStarted(barrier);
   507	
   508	    // Set callback to be notified when worker emits started()
   509	    job->setTestWorkerStartedCallback([workerStartedSem]() {
   510	        workerStartedSem->release();
   511	    });
   512	
   513	    TestScope scope(job);
   514	    scope.addBarrier(barrier);
   515	
   516	    QSignalSpy spy(job, &KJob::result);
   517	    job->start();
   518	
   519	    // Wait for worker to emit started() and block at barrier
   520	    // This proves worker thread exists and has reached the first checkpoint
   521	    QVERIFY(tryAcquire(workerStartedSem, 5000));
   522	
   523	    int handle = job->extractionHandle();
   524	    QVERIFY(handle != -1);
   525	    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 1);
   526	
   527	    // Request cancellation while worker is blocked at barrier
   528	    // Worker has been created, thread started, but doWork() not yet executed past first checkpoint
   529	    job->kill();
   530	
   531	    // Release barrier to allow worker to proceed

+// Shared ownership for test synchronization objects
+// Ensures barriers and callbacks remain alive as long as any worker could access them
+using SemaphorePtr = std::shared_ptr<QSemaphore>;
+
+// Robust scope guard for test cleanup
+// Ensures barriers are released, cancellation requested, and worker/manager completion waited for
+// CRITICAL: Does NOT time out. Waits indefinitely (with event processing) to ensure
+// all workers complete before test resources (payload files, jobs, semaphores) are destroyed.
+// TestScope is destroyed first (declared last), so all test resources are still alive.
+// If a worker is truly stuck, the test will hang - this is correct behavior for a stuck test.
+struct TestScope {
+    explicit TestScope(ArchiveExtractionJob* job = nullptr)
+        : m_job(job)
+        , m_manager(&ArchiveExtractionManager::instance())
+    {}
+
+    ~TestScope() {
+        // 1. Release all barriers to unblock any waiting worker
+        for (auto& s : m_barriers) {
+            if (s) s->release();
+        }
+
+        // 2. Request cancellation if job still exists
+        // We check if job has no error (i.e., hasn't finished with an error or success)
+        if (m_job && m_job->error() == KJob::NoError) {
+            m_job->kill();
+        }
+
+        // 3. Wait for manager to clean up all entries
+        // CRITICAL: NO TIMEOUT. Wait indefinitely (with event processing) to ensure
+        // all workers complete before test resources (payload files, jobs, semaphores) are destroyed.
+        // TestScope is destroyed first (declared last), so all test resources are still alive.
+        // If a worker is truly stuck, the test will hang - this is correct behavior for a stuck test.
+        while (!m_manager->waitForAllFinished(1000)) {
+            QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
+        }
+
+        // 4. Process any remaining events
+        QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
+    }
+
+    // Register a barrier that should be released on cleanup
+    void addBarrier(SemaphorePtr barrier) {
+        if (barrier) m_barriers.push_back(std::move(barrier));
+    }
+
+    // Register the job under test
+    void setJob(ArchiveExtractionJob* job) {
+        m_job = job;
+    }
+
+private:
+    QPointer<ArchiveExtractionJob> m_job;
+    ArchiveExtractionManager* m_manager = nullptr;
+    std::vector<SemaphorePtr> m_barriers;
+};

+void ArchiveExtractionJobTest::testMultipleConcurrentJobs()
+{
+    // Test multiple concurrent jobs - entries remain independent and are each cleaned exactly once
+    // Ensure clean state from previous tests
+    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(5000));
+    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 0);
+
+    const int numJobs = 4;
+    std::vector<ArchiveExtractionJob*> jobs;
+    std::vector<std::unique_ptr<QSignalSpy>> spies;
+    std::vector<QString> payloadFiles;
+    std::vector<int> handles;
+
+    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation) + QLatin1String("/demo/");
+    QDir().mkpath(installDir);
+
+    // Use ScopedTempFile for deterministic ownership - declare BEFORE jobParent
+    // so tempArchives is destroyed AFTER jobParent (reverse order of declaration)
+    std::vector<ScopedTempFile> tempArchives;
+    tempArchives.reserve(numJobs);
+
+    // Parent object for automatic job cleanup
+    QObject jobParent;
+
+    std::vector<TestScope> scopes;
+    scopes.reserve(numJobs);
+
+    // Create jobs with both shared and distinct destinations
+    for (int i = 0; i < numJobs; ++i) {
+        tempArchives.emplace_back();
+        QVERIFY(tempArchives[i].open());
+        QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
+        QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
+        tempArchives[i]->write(sourceArchive.readAll());
+        tempArchives[i]->flush();
+        sourceArchive.close();
+        QString payloadFile = tempArchives[i].fileName();
+        payloadFiles.push_back(payloadFile);
+
+        // Alternate between same destination and distinct destinations
+        QString jobInstallDir = installDir;
+        if (i % 2 == 1) {
+            jobInstallDir = installDir + QStringLiteral("subdir%1/").arg(i);
+            QDir().mkpath(jobInstallDir);
+        }
+
+        ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, jobInstallDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive, &jobParent);
+        auto spy = std::make_unique<QSignalSpy>(job, &KJob::result);
+        jobs.push_back(job);
+        spies.push_back(std::move(spy));
+
+        // Set up scope guard for each job
+        scopes.emplace_back(job);
+    }
+
+    // Start all jobs
+    for (int i = 0; i < numJobs; ++i) {
+        jobs[i]->start();
+    }
+
+    // Collect handles
+    for (int i = 0; i < numJobs; ++i) {
+        int handle = jobs[i]->extractionHandle();
+        QVERIFY(handle != -1);
+        handles.push_back(handle);
+    }
+
+    // Verify all entries are active
+    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), numJobs);
+
+    // Wait for all jobs to complete - use collective event-driven wait to avoid
+    // race where a job finishes before its individual wait() begins.
+    QElapsedTimer waitTimer;
+    waitTimer.start();
+    const int waitTimeout = 30000;
+    while (true) {
+        bool allDone = true;
+        for (int i = 0; i < numJobs; ++i) {
+            if (spies[i]->count() < 1) {
+                allDone = false;
+                break;
+            }
+        }
+        if (allDone) {
+            break;
+        }
+        if (waitTimer.elapsed() >= waitTimeout) {
+            // Report which jobs didn't complete
+            for (int i = 0; i < numJobs; ++i) {
+                if (spies[i]->count() < 1) {
+                    QFAIL(qPrintable(QString("Job %1 did not complete within timeout").arg(i)));
+                }
+            }
+            break;
+        }
+        QCoreApplication::processEvents(QEventLoop::AllEvents, 50);
+    }
+
+    // Verify all jobs completed exactly once
+    for (int i = 0; i < numJobs; ++i) {
+        QCOMPARE(spies[i]->count(), 1);
+        QVERIFY(!jobs[i]->error());
+    }
+
+    // Manager entry count should return to zero
+    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(30000));
+    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 0);
+
+    // All payloads should be cleaned up
+    for (const QString &pf : payloadFiles) {
+        QVERIFY(!QFileInfo::exists(pf));
+    }
+
+    // Scopes automatically handle cleanup - jobs are deleted via jobParent,
+    // spies are auto-deleted via unique_ptr, tempArchives destroyed after jobParent
