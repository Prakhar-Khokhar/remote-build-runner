    32	// Shared ownership for test synchronization objects
    33	// Ensures barriers and callbacks remain alive as long as any worker could access them
    34	using SemaphorePtr = std::shared_ptr<QSemaphore>;
    35	
    36	// Robust scope guard for test cleanup
    37	// Ensures barriers are released, cancellation requested, and worker/manager completion waited for
    38	struct TestScope {
    39	    explicit TestScope(ArchiveExtractionJob* job = nullptr)
    40	        : m_job(job)
    41	        , m_manager(&ArchiveExtractionManager::instance())
    42	    {}
    43	
    44	    ~TestScope() {
    45	        // 1. Release all barriers to unblock any waiting worker
    46	        for (auto& s : m_barriers) {
    47	            if (s) s->release();
    48	        }
    49	
    50	        // 2. Request cancellation if job still exists
    51	        // We check if job has no error (i.e., hasn't finished with an error or success)
    52	        if (m_job && m_job->error() == KJob::NoError) {
    53	            m_job->kill();
    54	        }
    55	
    56	        // 3. Wait for manager to clean up all entries
    57	        // Use generous timeout since we're in cleanup
    58	        const int cleanupTimeoutMs = 10000;
    59	        const bool cleaned = m_manager->waitForAllFinished(cleanupTimeoutMs);
    60	        if (!cleaned) {
    61	            qWarning() << "TestScope: Manager cleanup timed out after" << cleanupTimeoutMs << "ms";
    62	        }
    63	
    64	        // 4. Process any remaining events
    65	        QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
    66	    }
    67	
    68	    // Register a barrier that should be released on cleanup
    69	    void addBarrier(SemaphorePtr barrier) {
    70	        if (barrier) m_barriers.push_back(std::move(barrier));
    71	    }
    72	
    73	    // Register the job under test
    74	    void setJob(ArchiveExtractionJob* job) {
    75	        m_job = job;
    76	    }
    77	
    78	private:
    79	    QPointer<ArchiveExtractionJob> m_job;
    80	    ArchiveExtractionManager* m_manager = nullptr;
    81	    std::vector<SemaphorePtr> m_barriers;
    82	};

    85	inline SemaphorePtr makeSemaphore(int initial = 0) {
    86	    return std::make_shared<QSemaphore>(initial);

    89	// Safe wrapper for waiting on a semaphore with timeout
    90	// Returns true if acquired, false on timeout
    91	inline bool tryAcquire(SemaphorePtr sem, int timeoutMs = 5000) {
    92	    return sem && sem->tryAcquire(1, timeoutMs);
    93	}

    96	inline void release(SemaphorePtr sem) {
    97	    if (sem) sem->release();

   189	void ArchiveExtractionJobTest::cleanup()
   190	{
   191	    // Wait for any pending extractions to complete before next test
   192	    // This ensures clean state between tests since manager is a singleton
   193	    const bool cleaned = ArchiveExtractionManager::instance().waitForAllFinished(5000);
   194	    if (!cleaned) {
   195	        qWarning() << "cleanup(): Manager did not finish all extractions within timeout";
   196	        QFAIL("Manager cleanup timed out between tests - dangling extractions may contaminate next test");
   197	    }
   198	
   199	    // Do not clear destination mutexes here - mutexes are process-lifetime and
   200	    // clearing them while worker threads might still be running is unsafe.
   201	    // Mutexes will be cleaned up at process exit.
   202	}

   480	void ArchiveExtractionJobTest::testCancellationAfterWorkerCreationBeforeExecution()
   481	{
   482	    // Test cancellation after worker thread created but before worker executes doWork()
   483	    // Uses barrier to deterministically block worker at startup checkpoint
   484	    auto barrier = makeSemaphore();
   485	    auto workerStartedSem = makeSemaphore();
   486	    ScopedTempFile tempArchive;
   487	    QVERIFY(tempArchive.open());
   488	    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
   489	    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
   490	    tempArchive->write(sourceArchive.readAll());
   491	    tempArchive->flush();
   492	    sourceArchive.close();
   493	    QString payloadFile = tempArchive.fileName();
   494	
   495	    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation) + QLatin1String("/demo/");
   496	    QDir().mkpath(installDir);
   497	
   498	    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive);
   499	
   500	    // Configure worker to block at the barrier after started() is emitted
   501	    job->setTestBarrierAfterStarted(barrier);
   502	
   503	    // Set callback to be notified when worker emits started()
   504	    job->setTestWorkerStartedCallback([workerStartedSem]() {
   505	        workerStartedSem->release();
   506	    });
   507	
   508	    TestScope scope(job);
   509	    scope.addBarrier(barrier);
   510	
   511	    QSignalSpy spy(job, &KJob::result);
   512	    job->start();
   513	
   514	    // Wait for worker to emit started() and block at barrier
   515	    // This proves worker thread exists and has reached the first checkpoint
   516	    QVERIFY(tryAcquire(workerStartedSem, 5000));
   517	
   518	    int handle = job->extractionHandle();
   519	    QVERIFY(handle != -1);
   520	    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 1);
   521	
   522	    // Request cancellation while worker is blocked at barrier
   523	    // Worker has been created, thread started, but doWork() not yet executed past first checkpoint
   524	    job->kill();
   525	
   526	    // Release barrier to allow worker to proceed
   527	    release(barrier);
   528	
   529	    QVERIFY(spy.wait(5000));
   530	    // Should report KilledJobError
   531	    QVERIFY(job->error() == KJob::KilledJobError);
   532	    QVERIFY(job->errorText().contains("cancelled", Qt::CaseInsensitive));
   533	
   534	    // Payload should be removed on Cancelled
   535	    QVERIFY(!QFileInfo::exists(payloadFile));
   536

+// Ensures barriers and callbacks remain alive as long as any worker could access them
+using SemaphorePtr = std::shared_ptr<QSemaphore>;

+// Robust scope guard for test cleanup
+// Ensures barriers are released, cancellation requested, and worker/manager completion waited for
+struct TestScope {
+    explicit TestScope(ArchiveExtractionJob* job = nullptr)
+        : m_job(job)
+        , m_manager(&ArchiveExtractionManager::instance())
+    {}
+
+    ~TestScope() {
+        // 1. Release all barriers to unblock any waiting worker
+        for (auto& s : m_barriers) {
+            if (s) s->release();
+        }
+
+        // 2. Request cancellation if job still exists
+        // We check if job has no error (i.e., hasn't finished with an error or success)
+        if (m_job && m_job->error() == KJob::NoError) {
+            m_job->kill();
+        }
+
+        // 3. Wait for manager to clean up all entries
+        // Use generous timeout since we're in cleanup
+        const int cleanupTimeoutMs = 10000;
+        const bool cleaned = m_manager->waitForAllFinished(cleanupTimeoutMs);
+        if (!cleaned) {
+            qWarning() << "TestScope: Manager cleanup timed out after" << cleanupTimeoutMs << "ms";
+        }
+
+        // 4. Process any remaining events
+        QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
+    }
+
+    // Register a barrier that should be released on cleanup
+    void addBarrier(SemaphorePtr barrier) {
+        if (barrier) m_barriers.push_back(std::move(barrier));
+    }
+
+    // Register the job under test
+    void setJob(ArchiveExtractionJob* job) {
+        m_job = job;
+    }
+
+private:
+    QPointer<ArchiveExtractionJob> m_job;
+    ArchiveExtractionManager* m_manager = nullptr;
+    std::vector<SemaphorePtr> m_barriers;
+};

+void ArchiveExtractionJobTest::testMultipleConcurrentJobs()
+{
+    // Test multiple concurrent jobs - entries remain independent and are each cleaned exactly once
+    // Ensure clean state from previous tests
+    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(5000));
+    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 0);
+
+    const int numJobs = 4;
+    std::vector<ArchiveExtractionJob*> jobs;
+    std::vector<std::unique_ptr<QSignalSpy>> spies;
+    std::vector<QString> payloadFiles;
+    std::vector<int> handles;
+
+    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation) + QLatin1String("/demo/");
+    QDir().mkpath(installDir);
+
+    // Use ScopedTempFile for deterministic ownership - declare BEFORE jobParent
+    // so tempArchives is destroyed AFTER jobParent (reverse order of declaration)
+    std::vector<ScopedTempFile> tempArchives;
+    tempArchives.reserve(numJobs);
+
+    // Parent object for automatic job cleanup
+    QObject jobParent;
+
+    std::vector<TestScope> scopes;
+    scopes.reserve(numJobs);
+
+    // Create jobs with both shared and distinct destinations
+    for (int i = 0; i < numJobs; ++i) {
+        tempArchives.emplace_back();
+        QVERIFY(tempArchives[i].open());
+        QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
+        QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
+        tempArchives[i]->write(sourceArchive.readAll());
+        tempArchives[i]->flush();
+        sourceArchive.close();
+        QString payloadFile = tempArchives[i].fileName();
+        payloadFiles.push_back(payloadFile);
+
+        // Alternate between same destination and distinct destinations
+        QString jobInstallDir = installDir;
+        if (i % 2 == 1) {
+            jobInstallDir = installDir + QStringLiteral("subdir%1/").arg(i);
+            QDir().mkpath(jobInstallDir);
+        }
+
+        ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, jobInstallDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive, &jobParent);
+        auto spy = std::make_unique<QSignalSpy>(job, &KJob::result);
+        jobs.push_back(job);
+        spies.push_back(std::move(spy));
+
+        // Set up scope guard for each job
+        scopes.emplace_back(job);
+    }
+
+    // Start all jobs
+    for (int i = 0; i < numJobs; ++i) {
+        jobs[i]->start();
+    }
+
+    // Collect handles
+    for (int i = 0; i < numJobs; ++i) {
+        int handle = jobs[i]->extractionHandle();
+        QVERIFY(handle != -1);
+        handles.push_back(handle);
+    }
+
+    // Verify all entries are active
+    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), numJobs);
+
+    // Wait for all jobs to complete - use collective event-driven wait to avoid
+    // race where a job finishes before its individual wait() begins.
+    QElapsedTimer waitTimer;
+    waitTimer.start();
+    const int waitTimeout = 30000;
+    while (true) {
+        bool allDone = true;
+        for (int i = 0; i < numJobs; ++i) {
+            if (spies[i]->count() < 1) {
+                allDone = false;
+                break;
+            }
+        }
+        if (allDone) {
+            break;
+        }
+        if (waitTimer.elapsed() >= waitTimeout) {
+            // Report which jobs didn't complete
+            for (int i = 0; i < numJobs; ++i) {
+                if (spies[i]->count() < 1) {
+                    QFAIL(qPrintable(QString("Job %1 did not complete within timeout").arg(i)));
+                }
+            }
+            break;
+        }
+        QCoreApplication::processEvents(QEventLoop::AllEvents, 50);
+    }
+
+    // Verify all jobs completed exactly once
+    for (int i = 0; i < numJobs; ++i) {
+        QCOMPARE(spies[i]->count(), 1);
+        QVERIFY(!jobs[i]->error());
+    }
+
+    // Manager entry count should return to zero
+    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(30000));
+    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 0);
+
+    // All payloads should be cleaned up
+    for (const QString &pf : payloadFiles) {
+        QVERIFY(!QFileInfo::exists(pf));
+    }

+    // spies are auto-deleted via unique_ptr, tempArchives destroyed after jobParent
