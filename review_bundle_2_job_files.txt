    QString standardResourceDirectory;

    // Cancellation flag shared with manager entry
    std::shared_ptr<std::atomic<bool>> cancellation;
};

ArchiveExtractionJob::ArchiveExtractionJob(const QString &payloadFile, const QString &installDir, UncompressionOption uncompressionOption, QObject *parent)
    : KJob(parent)
    , d(new ArchiveExtractionJobPrivate)
{
    d->payloadFile = payloadFile;
    d->installDir = installDir;
    d->uncompressionOption = uncompressionOption;
    d->cancellation = std::make_shared<std::atomic<bool>>(false);
}

ArchiveExtractionJob::~ArchiveExtractionJob()
{
    // If kill was requested but start was never called, clean up the payload
    if (d->killRequested && !d->startCalled) {
        QFile::remove(d->payloadFile);
    }

    // If the job hasn't finished yet (destroyed while worker is running),
    // request cancellation. If the job already finished, the manager has
    // already cleaned up via QThread::finished → reapEntry.
    if (d->extractionHandle != -1 && !d->finished) {
        d->cancellation->store(true, std::memory_order_release);
        ArchiveExtractionManager::instance().requestCancellation(d->extractionHandle);
    }
}

void ArchiveExtractionJob::start()
{
    if (d->startCalled) {
        return;
    }
    d->startCalled = true;

    // If kill was requested before start, finish immediately with KilledJobError
    // and clean up the payload since no worker will be created to handle it.
    if (d->killRequested) {
        QFile::remove(d->payloadFile);
        if (!error()) {
            setError(KJob::KilledJobError);
            setErrorText(i18n("Archive extraction was cancelled"));
            emitResult();
        } else if (error() == KJob::KilledJobError && errorText().isEmpty()) {
            setErrorText(i18n("Archive extraction was cancelled"));
        }
        return;
    }

    // Start extraction through the manager. The manager assigns the extraction
    // handle via setExtractionHandle() before any result delivery. We must not
    // access d after this call returns: a synchronous result handler may have
    // destroyed the job, freeing d.
    ArchiveExtractionManager::instance().startExtraction(
        d->payloadFile, d->installDir, d->uncompressionOption, this);
}

bool ArchiveExtractionJob::doKill()
{
    d->cancellation->store(true, std::memory_order_release);
    d->killRequested = true;

    // If already finished, nothing to do
    if (d->finished) {
        return true;
    }

    // If extraction hasn't started yet, we can finish immediately
    if (d->extractionHandle == -1) {
        return true;
    }

    // Request cancellation through manager
    ArchiveExtractionManager::instance().requestCancellation(d->extractionHandle);

    // Return false - job will finish when worker completes and emits result
    return false;
}

void ArchiveExtractionJob::handleSuccess(const QStringList &files)
{
    // Must be called from the job's owning thread (main thread)
    Q_ASSERT(QThread::currentThread() == thread());

    qCDebug(KNEWSTUFFCORE) << "Job handleSuccess called, finished=" << d->finished << "isFinished=" << isFinished();
    if (d->finished || isFinished()) {
        return;
    }
    d->finished = true;
    d->installedFiles = files;
    qCDebug(KNEWSTUFFCORE) << "Job emitting result signal";
    emitResult();
}

void ArchiveExtractionJob::handleFallbackRequired()
{
    // Must be called from the job's owning thread (main thread)
    Q_ASSERT(QThread::currentThread() == thread());

    if (d->finished || isFinished()) {
        return;
    }
    d->finished = true;
    setProperty("fallbackRequired", true);
    emitResult();
}

void ArchiveExtractionJob::handleExtractionFailed(const QString &errorMessage)
{
    // Must be called from the job's owning thread (main thread)
    Q_ASSERT(QThread::currentThread() == thread());

    if (d->finished || isFinished()) {
        return;
    }
    d->finished = true;
    setError(KJob::UserDefinedError);
    setErrorText(errorMessage);
    emitResult();
}

void ArchiveExtractionJob::handleCancelled()
{
    // Must be called from the job's owning thread (main thread)
    Q_ASSERT(QThread::currentThread() == thread());

    if (d->finished || isFinished()) {
        return;
    }
    d->finished = true;
    setError(KJob::KilledJobError);
    setErrorText(i18n("Archive extraction was cancelled"));
    emitResult();
}
