// Shared ownership for test synchronization objects
// Ensures barriers and callbacks remain alive as long as any worker could access them
using SemaphorePtr = std::shared_ptr<QSemaphore>;

// Thread-safe test state holder for callbacks that may execute asynchronously
// Uses shared_ptr + atomic to ensure lifetime extends beyond test scope
template<typename T>
struct TestState {
    std::shared_ptr<std::atomic<T>> value;
    TestState(T initial = T{}) : value(std::make_shared<std::atomic<T>>(initial)) {}
    T load() const { return value->load(std::memory_order_acquire); }
    void store(T v) { value->store(v, std::memory_order_release); }
    void increment() { value->fetch_add(1, std::memory_order_acq_rel); }
};

// Robust scope guard for test cleanup
// Ensures cancellation is requested, barriers are released, and worker/manager completion waited for
// CRITICAL: Bounded wait with safe termination strategy.
// - Requests cancellation FIRST if job is still active (so workers observe cancellation at next checkpoint)
// - Releases barriers AFTER cancellation requested (unblocks workers to hit their cancellation checkpoints)
// - Waits for manager cleanup with generous timeout
// - If timeout expires with active workers, calls std::quick_exit to avoid unsafe process exit
// - Never destroys a running QThread. Never frees worker-owned state while its thread may still be using it.
// - Never uses nested GUI event loops or unsafe forced deletion as a workaround.
// - Note: KArchiveDirectory::copyTo() is non-interruptible; a worker blocked in copyTo() will not see
//   cancellation until copyTo() returns. This is an inherent limitation of the archive library.
struct TestScope {
    explicit TestScope(ArchiveExtractionJob* job = nullptr)
        : m_job(job)
        , m_manager(&ArchiveExtractionManager::instance())
    {}

    // Non-copyable, non-movable: owns barriers and manages cleanup; copying would duplicate
    // barrier ownership and cause double-release or missed cleanup.
    TestScope(const TestScope&) = delete;
    TestScope& operator=(const TestScope&) = delete;
    TestScope(TestScope&&) = delete;
    TestScope& operator=(TestScope&&) = delete;

    ~TestScope() {
        // 1. Request cancellation FIRST if job still exists and hasn't finished.
        // This ensures that when barriers are released, workers will observe
        // the cancellation flag at their next checkpoint.
        if (m_job && m_job->error() == KJob::NoError) {
            m_job->kill();
        }

        // 2. Release all barriers to unblock any waiting worker at their checkpoints.
        // Ordering: cancellation requested FIRST, then barriers released.
        // Workers resume and immediately hit their next cancellation checkpoint.
        for (auto& s : m_barriers) {
            if (s) s->release();
        }

        // 3. Wait for manager to clean up all entries with bounded timeout
        constexpr int cleanupTimeoutMs = 30000; // Generous timeout for worker completion + thread shutdown
        QElapsedTimer timer;
        timer.start();
        while (!m_manager->waitForAllFinished(1000)) {
            QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
            if (timer.elapsed() >= cleanupTimeoutMs) {
                // Timeout: emit diagnostic with manager/entry state
                qCritical() << "TestScope cleanup timeout after" << cleanupTimeoutMs << "ms";
                qCritical() << "Active extractions count:" << m_manager->activeExtractionsCount();
                qCritical() << "Worker threads may still be running. Terminating test process to avoid unsafe exit.";
                
                // Use std::quick_exit to terminate without destroying C++ objects
                // (which could race with still-running worker threads).
                // This is the safest option when threads cannot be joined.
                std::quick_exit(EXIT_FAILURE);
            }
        }

        // 4. Process any remaining events
        QCoreApplication::processEvents(QEventLoop::AllEvents, 100);
    }

    // Register a barrier that should be released on cleanup
    void addBarrier(SemaphorePtr barrier) {
        if (barrier) m_barriers.push_back(std::move(barrier));
    }

    // Register the job under test
    void setJob(ArchiveExtractionJob* job) {
        m_job = job;
    }

private:
    QPointer<ArchiveExtractionJob> m_job;
    ArchiveExtractionManager* m_manager = nullptr;
    std::vector<SemaphorePtr> m_barriers;
};

// Helper to create a shared semaphore
inline SemaphorePtr makeSemaphore(int initial = 0) {
    return std::make_shared<QSemaphore>(initial);
}

// Safe wrapper for waiting on a semaphore with timeout
// Returns true if acquired, false on timeout
inline bool tryAcquire(SemaphorePtr sem, int timeoutMs = 5000) {
    return sem && sem->tryAcquire(1, timeoutMs);
}

// Helper to release a shared semaphore
inline void release(SemaphorePtr sem) {
    if (sem) sem->release();
}

// Test-scoped temporary file that auto-removes on scope exit
    // Ensures cleanup even on test failure
    struct ScopedTempFile {
        explicit ScopedTempFile(const QString& templateName = QString())
            : m_file(std::make_unique<QTemporaryFile>(templateName)) {
            m_file->setAutoRemove(true);
        }

        // Move-only type
        ScopedTempFile(ScopedTempFile&& other) noexcept = default;
        ScopedTempFile& operator=(ScopedTempFile&& other) noexcept = default;
        ScopedTempFile(const ScopedTempFile&) = delete;
        ScopedTempFile& operator=(const ScopedTempFile&) = delete;

        bool open() { return m_file->open(); }
        QString fileName() const { return m_file->fileName(); }
        QTemporaryFile* operator->() { return m_file.get(); }
        QTemporaryFile& operator*() { return *m_file; }

    private:
        std::unique_ptr<QTemporaryFile> m_file;
    };

    // Do not clear destination mutexes here - mutexes are process-lifetime and

    // Payload should be removed on Success

    job->deleteLater();
    QPointer<ArchiveExtractionJob> jobPtr(job);
    QTRY_VERIFY(jobPtr.isNull());

void ArchiveExtractionJobTest::testCancellationAfterWorkerCreationBeforeExecution()
{
    // Test cancellation after worker thread created but before worker executes doWork()
    // Uses barrier to deterministically block worker at startup checkpoint
    auto barrier = makeSemaphore();
    auto workerStartedSem = makeSemaphore();
    ScopedTempFile tempArchive;
    QVERIFY(tempArchive.open());
    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
    tempArchive->write(sourceArchive.readAll());
    tempArchive->flush();
    sourceArchive.close();
    QString payloadFile = tempArchive.fileName();

    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation) + QLatin1String("/demo/");
    QDir().mkpath(installDir);

    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive);

    // Configure worker to block at the barrier after started() is emitted
    job->setTestBarrierAfterStarted(barrier);

    // Set callback to be notified when worker emits started()
    job->setTestWorkerStartedCallback([workerStartedSem]() {
        workerStartedSem->release();
    });

    TestScope scope(job);
    scope.addBarrier(barrier);

    QSignalSpy spy(job, &KJob::result);
    job->start();

    // Wait for worker to emit started() and block at barrier
    // This proves worker thread exists and has reached the first checkpoint
    QVERIFY(tryAcquire(workerStartedSem, 5000));

    int handle = job->extractionHandle();
    QVERIFY(handle != -1);
    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 1);

    // Request cancellation while worker is blocked at barrier
    // Worker has been created, thread started, but doWork() not yet executed past first checkpoint
    job->kill();

    // Release barrier to allow worker to proceed
    release(barrier);

    QVERIFY(spy.wait(5000));
    // Should report KilledJobError
    QVERIFY(job->error() == KJob::KilledJobError);
    QVERIFY(job->errorText().contains("cancelled", Qt::CaseInsensitive));

    job->deleteLater();
    QPointer<ArchiveExtractionJob> jobPtr(job);
    QTRY_VERIFY(jobPtr.isNull());
}

void ArchiveExtractionJobTest::testCancellationBeforeCopyTo()
{
    // Test cancellation before copyTo - worker blocked at the barrier before the new
    // pre-copyTo cancellation checkpoint. Proves the new checkpoint is effective.
    // Uses a unique destination directory so previous test output cannot affect the assertion.
    auto barrier = makeSemaphore();
    auto workerAtBarrierSem = makeSemaphore();
    TestState<bool> preCopyToReached(false);

    ScopedTempFile tempArchive;
    QVERIFY(tempArchive.open());
    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
    tempArchive->write(sourceArchive.readAll());
    tempArchive->flush();
    sourceArchive.close();
    QString payloadFile = tempArchive.fileName();

    // Use a unique destination directory for this test
    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation) + QLatin1String("/demo_testCancellationBeforeCopyTo/");
    QDir().mkpath(installDir);
    // Ensure destination is empty at start
    QVERIFY(QDir(installDir).entryList(QDir::Files | QDir::Dirs | QDir::NoDotAndDotDot).isEmpty());

    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive);

    // Configure worker to block at the barrier before copyTo (after aboutToCopyTo, before checkpoint)
    job->setTestBarrierBeforeCopyTo(barrier);

    // Callback fires when worker reaches the barrier (before the new checkpoint)
    job->setTestBeforeCopyToCallback([workerAtBarrierSem]() {
        workerAtBarrierSem->release();
    });

    // Observation point: fires AFTER the new pre-copyTo cancellation checkpoint but BEFORE copyTo()
    // If this fires, the checkpoint was passed (cancellation not observed at checkpoint).
    // If cancellation is requested while worker is blocked at barrier, the checkpoint should
    // catch it and this callback should NEVER fire.
    job->setTestPreCopyToCallback([preCopyToReached]() mutable {
        preCopyToReached.store(true);
    });

    TestScope scope(job);
    scope.addBarrier(barrier);

    QSignalSpy spy(job, &KJob::result);
    job->start();

    // Wait for worker to reach the barrier (proves: MIME checked, archive opened, mutex acquired, aboutToCopyTo emitted)
    QVERIFY(tryAcquire(workerAtBarrierSem, 5000));

    // Request cancellation while worker is blocked at the barrier, BEFORE the new checkpoint
    job->kill();

    // Release barrier to allow worker to proceed
    // Worker will hit the new cancellation checkpoint after barrier and before copyTo()
    release(barrier);

    QVERIFY(spy.wait(5000));
    // Should report KilledJobError
    QVERIFY(job->error() == KJob::KilledJobError);
    QVERIFY(job->errorText().contains("cancelled", Qt::CaseInsensitive));

    // The pre-copyTo observation point must NOT have been reached - proves the new checkpoint caught cancellation
    QVERIFY(!preCopyToReached.load());

    // Payload should be removed on Cancelled
    QVERIFY(!QFileInfo::exists(payloadFile));

    // Destination must remain empty - no partial extraction occurred
    QVERIFY(QDir(installDir).entryList(QDir::Files | QDir::Dirs | QDir::NoDotAndDotDot).isEmpty());

    job->deleteLater();
    QPointer<ArchiveExtractionJob> jobPtr(job);
    QTRY_VERIFY(jobPtr.isNull());
}

void ArchiveExtractionJobTest::testCancellationBeforeCancellationCheck()
{
    // Test cancellation at the first pre-copyTo cancellation check (after mutex acquisition)
    // Uses a dedicated barrier that fires after mutex acquisition but BEFORE the cancellation check

void ArchiveExtractionJobTest::testManagerMediatedCancellation()
{
    // Test manager-mediated cancellation via ArchiveExtractionManager::requestCancellation()
    // without calling job->kill() and without destroying the job.
    // This verifies the manager's atomic cancellation store works independently of the
    // queued interruption fallback. The test suppresses the fallback to isolate the atomic store.
    auto barrier = makeSemaphore();
    auto workerAtBarrierSem = makeSemaphore();
    TestState<bool> preCopyToReached(false);

    ScopedTempFile tempArchive;
    QVERIFY(tempArchive.open());
    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
    tempArchive->write(sourceArchive.readAll());
    tempArchive->flush();
    sourceArchive.close();
    QString payloadFile = tempArchive.fileName();

    // Use a unique destination directory for this test
    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation) + QLatin1String("/demo_testManagerMediatedCancellation/");

    // Ensure destination is empty at start
    QVERIFY(QDir(installDir).entryList(QDir::Files | QDir::Dirs | QDir::NoDotAndDotDot).isEmpty());

    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir, ArchiveExtractionJob::UncompressIntoSubdirIfArchive);

    // Configure worker to block at the barrier before copyTo
    job->setTestBarrierBeforeCopyTo(barrier);

    // Callback fires when worker reaches the barrier (before the new checkpoint)
    job->setTestBeforeCopyToCallback([workerAtBarrierSem]() {
        workerAtBarrierSem->release();
    });

    // Observation point: fires AFTER the pre-copyTo cancellation checkpoint but BEFORE copyTo()
    // If this fires, the checkpoint was passed (cancellation not observed at checkpoint).
    job->setTestPreCopyToCallback([preCopyToReached]() mutable {
        preCopyToReached.store(true);
    });

    // Capture worker's terminal result via test-only callback
    TestState<ArchiveExtractionWorker::ResultType> observedWorkerResult(ArchiveExtractionWorker::Success);
    job->setTestWorkerResultCallback([observedWorkerResult](ArchiveExtractionWorker::ResultType result) mutable {
        observedWorkerResult.store(result);
    });

    TestScope scope(job);
    scope.addBarrier(barrier);

    QSignalSpy resultSpy(job, &KJob::result);
    job->start();

    // Wait for worker to reach the barrier (proves: MIME checked, archive opened, mutex acquired, aboutToCopyTo emitted)
    QVERIFY(tryAcquire(workerAtBarrierSem, 5000));

    int handle = job->extractionHandle();
    QVERIFY(handle != -1);
    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 1);

    // Suppress the queued interruption fallback to isolate the atomic cancellation store
    // This is the key test seam: with this suppressed, ONLY the atomic store can cancel the worker
    ArchiveExtractionManager::instance().setTestSuppressInterruptionFallback(handle, true);

    // This tests the manager's cancellation API in isolation
    ArchiveExtractionManager::instance().requestCancellation(handle);

    // With interruption fallback suppressed, ONLY the atomic flag can trigger cancellation

    QVERIFY(job->error() == KJob::KilledJobError);
    QVERIFY(job->errorText().contains("cancelled", Qt::CaseInsensitive));

    // Verify worker's actual terminal result was Cancelled
    QCOMPARE(observedWorkerResult.load(), ArchiveExtractionWorker::Cancelled);

    // The pre-copyTo observation point must NOT have been reached - proves the atomic checkpoint caught cancellation
    QVERIFY(!preCopyToReached.load());

    // Payload should be removed on Cancelled
    QVERIFY(!QFileInfo::exists(payloadFile));

    // Destination must remain empty - no partial extraction occurred
    QVERIFY(QDir(installDir).entryList(QDir::Files | QDir::Dirs | QDir::NoDotAndDotDot).isEmpty());

    job->deleteLater();

    }

    // Start all jobs
    for (int i = 0; i < numJobs; ++i) {
        jobs[i]->start();
    }

    // Collect handles
    for (int i = 0; i < numJobs; ++i) {
        int handle = jobs[i]->extractionHandle();
        QVERIFY(handle != -1);
        handles.push_back(handle);
    }

    // Verify all entries are active
    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), numJobs);

    // Wait for all jobs to complete - use collective event-driven wait to avoid
    // race where a job finishes before its individual wait() begins.
    QElapsedTimer waitTimer;
    waitTimer.start();
    const int waitTimeout = 30000;
    while (true) {
        bool allDone = true;
        for (int i = 0; i < numJobs; ++i) {
            if (spies[i]->count() < 1) {
                allDone = false;
                break;
            }
        }
        if (allDone) {
            break;
        }
        if (waitTimer.elapsed() >= waitTimeout) {
            // Report which jobs didn't complete
            for (int i = 0; i < numJobs; ++i) {
                if (spies[i]->count() < 1) {
                    QFAIL(qPrintable(QString("Job %1 did not complete within timeout").arg(i)));
                }
            }
            break;
        }
        QCoreApplication::processEvents(QEventLoop::AllEvents, 50);
    }

    // Verify all jobs completed exactly once
    for (int i = 0; i < numJobs; ++i) {
        QCOMPARE(spies[i]->count(), 1);
        QVERIFY(!jobs[i]->error());
    }

    // Manager entry count should return to zero
    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(30000));
    QCOMPARE(ArchiveExtractionManager::instance().activeExtractionsCount(), 0);

    // All payloads should be cleaned up
    for (const QString &pf : payloadFiles) {
        QVERIFY(!QFileInfo::exists(pf));
    }

    // Scopes automatically handle cleanup - jobs are deleted via jobParent,
    // spies are auto-deleted via unique_ptr, tempArchives destroyed after jobParent
}

void ArchiveExtractionJobTest::testSyntheticFailureSynchronousJobDestruction()
{
    // C4: Prove that the synthetic-failure path cannot use the job after a
    // synchronous result handler destroys it. The job must have a valid
    // handle/state before the result callback runs, and startExtraction()
    // must return safely even if the job is destroyed during result delivery.
    ArchiveExtractionManager::instance().setTestForceStartFailure(true);
    // RAII-style reset: ensure the seam is turned off even if the test fails.
    struct SeamResetter {
        ~SeamResetter() { ArchiveExtractionManager::instance().setTestForceStartFailure(false); }
    } seamResetter;
    Q_UNUSED(seamResetter);

    bool handlerRanDuringStartExtraction = false;
    bool resultReceived = false;
    int jobErrorCode = -1;

    ScopedTempFile tempArchive;
    QVERIFY(tempArchive.open());
    QFile sourceArchive(dataDir + QStringLiteral("archive_dir.tar.gz"));
    QVERIFY(sourceArchive.open(QIODevice::ReadOnly));
    tempArchive->write(sourceArchive.readAll());
    tempArchive->flush();
    sourceArchive.close();
    QString payloadFile = tempArchive.fileName();

    QString installDir = QStandardPaths::writableLocation(QStandardPaths::GenericDataLocation)
                         + QStringLiteral("/demo_testSyntheticFailureSync/");
    QDir().mkpath(installDir);

    ArchiveExtractionJob *job = new ArchiveExtractionJob(payloadFile, installDir,
                                                         ArchiveExtractionJob::UncompressIntoSubdirIfArchive);
    job->setAutoDelete(false);
    QPointer<ArchiveExtractionJob> jobPtr(job);

    // Connect a result handler that destroys the job via deleteLater() during
    // synchronous result delivery. The lambda sets handlerRanDuringStartExtraction
    // to prove the handler ran while startExtraction() was still executing
    // (i.e., before it returned to ArchiveExtractionJob::start()).
    QObject::connect(job, &KJob::result, job, [&handlerRanDuringStartExtraction,
                                                &resultReceived, &jobErrorCode,
                                                job](KJob *finishedJob) {
        Q_UNUSED(finishedJob);
        resultReceived = true;
        QVERIFY(job != nullptr);  // Same job object
        QVERIFY(job->extractionHandle() != -1);  // Handle valid before result
        jobErrorCode = job->error();
        handlerRanDuringStartExtraction = true;
        job->deleteLater();
    });

    // startExtraction() may synchronously emit a result, which runs the handler
    // above. The handler calls deleteLater() before startExtraction() returns.
    // After start() returns, d must not be accessed (C4 fix). Do NOT access job
    // here: the job may have been destroyed via deleteLater() during
    // startExtraction().
    job->start();

    // The result handler ran synchronously during startExtraction().
    QVERIFY(resultReceived);
    QVERIFY(handlerRanDuringStartExtraction);
    QCOMPARE(jobErrorCode, KJob::UserDefinedError);

    // Process events to let deleteLater() complete destruction.
    QTRY_VERIFY(jobPtr.isNull());  // Job was destroyed

    // Entry should be released (job was destroyed, tryFinalize removed it).
    QVERIFY(ArchiveExtractionManager::instance().waitForAllFinished(5000));
    QTRY_VERIFY(ArchiveExtractionManager::instance().testTotalEntryCount() == 0);

    // Payload removed by implementation (no worker was created).
    QVERIFY(!QFileInfo::exists(payloadFile));
}
